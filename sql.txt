-- BƯỚC 5: TRUY VẤN

-- 5.1. Hiển thị oID, oDate, oTotalPrice của tất cả hóa đơn
SELECT oID, oDate, oTotalPrice 
FROM `Order`;

-- 5.2. Danh sách khách hàng đã mua hàng + sản phẩm được mua
SELECT 
    c.cID,
    c.cName AS CustomerName,
    p.pName AS ProductName,
    od.odQTY AS Quantity
FROM Customer c
JOIN `Order` o ON c.cID = o.cID
JOIN OrderDetail od ON o.oID = od.oID
JOIN Product p ON od.pID = p.pID
ORDER BY c.cID, o.oID;

-- 5.3. Khách hàng KHÔNG mua bất kỳ sản phẩm nào (Anti-Join)
SELECT c.cID, c.cName
FROM Customer c
LEFT JOIN `Order` o ON c.cID = o.cID
WHERE o.oID IS NULL;

-- 5.4. Mã hóa đơn, ngày bán, giá tiền từng hóa đơn
SELECT 
    o.oID AS OrderID,
    o.oDate AS OrderDate,
    SUM(od.odQTY * p.pPrice) AS TotalPrice
FROM `Order` o
JOIN OrderDetail od ON o.oID = od.oID
JOIN Product p ON od.pID = p.pID
GROUP BY o.oID, o.oDate
ORDER BY o.oID;
